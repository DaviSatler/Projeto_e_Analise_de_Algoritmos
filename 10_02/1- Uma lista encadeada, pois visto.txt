1- Uma lista encadeada, pois visto termos pedidos assim como uma fila, teríamos que inserir no finall e também remover no início com uma certa frequência, sem deslocar elementos.


2- Eu implementaria essa lista por meio de uma array, pois a pesquisa binária necessita de acesso aleatório rápido aos elementos.


3-A principal desvantagem é que inserindo um novo usuário seria necessário manter o array ordenado e teria que deslocar vários elementos para abrir um espaço, o que torna a inserção custosa em tempo e processamento, principalmente quando a lista é grande.

4-A estrutura híbrida é mais eficiente comparada a de arrays e lista encadeadas isoladas, pois reduz o número de elementos percorridos ao dividir os dados em grupos. Inserções e remoções são mais rápidas que em arrays, e buscas são mais rápidas que em listas encadeadas únicas.


5- def pesquisa_linear(lista, alvo):
    for i in range(len(lista)):
        if lista[i] == alvo:
            return i
    return -1

lista = [3, 7, 1, 9, 5]
print(pesquisa_linear(lista, 9))


6- def pesquisa_binaria(lista, alvo):
    inicio = 0
    fim = len(lista) - 1

    while inicio <= fim:
        meio = (inicio + fim) // 2

        if lista[meio] == alvo:
            return meio
        elif lista[meio] < alvo:
            inicio = meio + 1
        else:
            fim = meio - 1

    return -1

lista = [1, 3, 5, 7, 9, 11, 15]
print(pesquisa_binaria(lista, 7))


7- def selection_sort(lista):
    n = len(lista)

    for i in range(n):
        menor = i

        for j in range(i + 1, n):
            if lista[j] < lista[menor]:
                menor = j

        lista[i], lista[menor] = lista[menor], lista[i]

    return lista

lista = [29, 10, 14, 37, 13]
print(selection_sort(lista))


8- def remover_duplicados(lista):
    nova = []

    for num in lista:
        if num not in nova:
            nova.append(num)

    return nova

lista = [4, 2, 9, 4, 3, 2, 8]
print(remover_duplicados(lista))


9- def intersecao(lista1, lista2):
    resultado = []

    for num in lista1:
        if num in lista2 and num not in resultado:
            resultado.append(num)

    return resultado

lista1 = [1, 3, 5, 7, 9]
lista2 = [3, 4, 5, 6, 9]
print(intersecao(lista1, lista2))

11- def mover_zeros(lista):
    nao_zeros = [x for x in lista if x != 0]
    zeros = [0] * lista.count(0)

    return nao_zeros + zeros

lista = [0, 1, 0, 3, 12]
print(mover_zeros(lista))
